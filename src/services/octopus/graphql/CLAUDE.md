- One file per API query
- Types generated by codegen
- @src/services/octopus/graphql/_constants.ts: Constants shared by all queries
- File structure:
```tsx
import { y } from "w";
import type { AbcQuery } from "#/graphql/graphql";

const STALE_TIME_MS = 5 * 60 * 1_000;

const QUERY = graphql(`
  query Abc($x: String!) {
    abc(x: $x) {
      a
      b
    }
  }
`);
export type AbcResp = AbcQuery;

export function useAbc(x: string | undefined) {
	const client = useQueryClient();
	const apiKey = useSettings().apiKey;

	return useQuery({
		queryKey: [OCTOPUS_QUERY_KEY, apiKey, "abc", x],
		queryFn:
			apiKey && x
				? async () => {
						const token = await fetchAuthToken(client, apiKey);
						return request(
							OCTOPUS_GRAPHQL_ENDPOINT,
							QUERY,
							{ x },
							{ Authorization: token },
						);
					}
				: skipToken,
		staleTime: STALE_TIME_MS,
	});
}
```
